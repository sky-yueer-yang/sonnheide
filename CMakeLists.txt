cmake_minimum_required(VERSION 3.20)
project(Sonnheide VERSION 0.9.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(SONN_SANITIZERS "Instrument authoritative core and application" OFF)
set(SONN_DEPS "${CMAKE_SOURCE_DIR}/.build/deps")
set(SONN_RUNTIME "${CMAKE_SOURCE_DIR}/.build/runtime")
include("${SONN_RUNTIME}/admitted_hashes.cmake")
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake")
find_package(Threads REQUIRED)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
add_subdirectory("${SONN_DEPS}/freetype" deps/freetype EXCLUDE_FROM_ALL)
set(BUILD_UNIVERSAL_BINARIES OFF CACHE BOOL "" FORCE)
set(BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(ENABLE_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SONN_DEPS}/rmlui" deps/rmlui EXCLUDE_FROM_ALL)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${SONN_DEPS}/sdl" deps/sdl EXCLUDE_FROM_ALL)
set(BX_DIR "${SONN_DEPS}/bx" CACHE STRING "" FORCE)
set(BIMG_DIR "${SONN_DEPS}/bimg" CACHE STRING "" FORCE)
set(BGFX_DIR "${SONN_DEPS}/bgfx" CACHE STRING "" FORCE)
set(BGFX_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BGFX_INSTALL OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_BIN2C OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_GEOMETRY OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_TEXTURE OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_EXAMPLE_COMMON OFF CACHE BOOL "" FORCE)
set(BGFX_CUSTOM_TARGETS OFF CACHE BOOL "" FORCE)
set(BGFX_CONFIG_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory("${SONN_DEPS}/bgfx-cmake" deps/bgfx EXCLUDE_FROM_ALL)
# Our admitted backends need HLSL/Metal, not WGSL; don't compile unused Tint.
get_target_property(_shader_links shaderc LINK_LIBRARIES)
list(FILTER _shader_links EXCLUDE REGEX "^(tint|webgpu)$")
set_property(TARGET shaderc PROPERTY LINK_LIBRARIES "${_shader_links}")
target_compile_definitions(shaderc PRIVATE SHADERC_CONFIG_HAS_TINT=0)
# No unadmitted AVIF decoder is compiled; preserve immutable upstream archives.
get_target_property(_decode_sources bimg_decode SOURCES)
list(FILTER _decode_sources EXCLUDE REGEX "(dav1d|libavif)")
set_property(TARGET bimg_decode PROPERTY SOURCES "${_decode_sources}")
target_compile_definitions(bimg_decode PRIVATE BIMG_CONFIG_PARSE_ENABLE=0 BIMG_CONFIG_PARSE_STB=1 BIMG_CONFIG_PARSE_JPEG=1 BIMG_CONFIG_PARSE_PNG=1 BIMG_CONFIG_PARSE_WIC=0)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS src/core/*.cpp)
add_library(sonn_core STATIC ${CORE_SOURCES})
target_include_directories(sonn_core PUBLIC src/core)
target_link_libraries(sonn_core PUBLIC Threads::Threads)
add_library(sonn_client STATIC src/client/client.cpp)
target_link_libraries(sonn_client PUBLIC sonn_core SDL3::SDL3-static RmlCore bgfx bx bimg_decode)
add_library(sonn_ui STATIC src/ui/ui.cpp src/ui/pixel_decorator.cpp)
target_link_libraries(sonn_ui PUBLIC RmlCore)
target_include_directories(sonn_ui PUBLIC src/ui)
add_executable(sonnheide apps/game/main.cpp)
target_link_libraries(sonnheide PRIVATE sonn_client sonn_ui)
target_compile_definitions(sonnheide PRIVATE SONN_SOURCE_ROOT="${CMAKE_SOURCE_DIR}" SONN_RESOURCE_ROOT="${SONN_RUNTIME}" SONN_DEFINITION_SHA="${SONN_DEFINITION_SHA}" SONN_ASSET_SHA="${SONN_ASSET_SHA}")
foreach(t sonn_core sonn_client sonn_ui sonnheide)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(SONN_SANITIZERS AND NOT MSVC)
    target_compile_options(${t} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PUBLIC -fsanitize=address,undefined)
  endif()
endforeach()
# RmlUi shares STL instantiations with the application. Matching instrumentation
# preserves libc++ container annotations instead of suppressing actual checks.
if(SONN_SANITIZERS AND NOT MSVC)
  target_compile_options(RmlCore PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(RmlCore PUBLIC -fsanitize=address,undefined)
endif()
if(APPLE)
  set(SHADER_PLATFORM osx)
  set(SHADER_BACKEND metal)
  set(SHADER_PROFILE metal)
elseif(WIN32)
  set(SHADER_PLATFORM windows)
  set(SHADER_BACKEND dx11)
else()
  message(FATAL_ERROR "Production GPU targets are Windows D3D11 and development Metal only")
endif()
set(SHADER_OUTPUTS)
foreach(pair world.vs world.fs water.fs sky.vs sky.fs ui.vs ui.fs shadow.vs shadow.fs)
  string(REGEX REPLACE ".*\\." "" stage "${pair}")
  if(stage STREQUAL "vs")
    set(SHADER_TYPE vertex)
    set(SHADER_PROFILE_WINDOWS s_5_0)
  else()
    set(SHADER_TYPE fragment)
    set(SHADER_PROFILE_WINDOWS s_5_0)
  endif()
  if(WIN32)
    set(SHADER_PROFILE ${SHADER_PROFILE_WINDOWS})
  endif()
  set(out "${SONN_RUNTIME}/shaders/${SHADER_BACKEND}/${pair}.bin")
  set(SURFACE_INCLUDE_DEPENDENCY)
  if(pair STREQUAL "world.fs" OR pair STREQUAL "water.fs")
    set(SURFACE_INCLUDE_DEPENDENCY assets/shaders/surface.sh)
  endif()
  add_custom_command(OUTPUT "${out}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${SONN_RUNTIME}/shaders/${SHADER_BACKEND}"
    COMMAND $<TARGET_FILE:shaderc> -f "${CMAKE_SOURCE_DIR}/assets/shaders/${pair}.sc" -o "${out}" --type ${SHADER_TYPE} --platform ${SHADER_PLATFORM} --profile ${SHADER_PROFILE} --varyingdef "${CMAKE_SOURCE_DIR}/assets/shaders/varying.def.sc" -i "${SONN_DEPS}/bgfx/src" -i "${CMAKE_SOURCE_DIR}/assets/shaders"
    DEPENDS shaderc "assets/shaders/${pair}.sc" assets/shaders/common.sh assets/shaders/varying.def.sc ${SURFACE_INCLUDE_DEPENDENCY} VERBATIM)
  list(APPEND SHADER_OUTPUTS "${out}")
endforeach()
add_custom_target(sonn_shaders ALL DEPENDS ${SHADER_OUTPUTS})
add_dependencies(sonnheide sonn_shaders)
enable_testing()
file(GLOB CORE_TESTS CONFIGURE_DEPENDS tests/core/*.cpp)
foreach(test_source ${CORE_TESTS})
  get_filename_component(test_name "${test_source}" NAME_WE)
  add_executable("${test_name}" "${test_source}")
  target_link_libraries("${test_name}" PRIVATE sonn_core)
  if(MSVC)
    target_compile_options("${test_name}" PRIVATE /utf-8)
  endif()
  target_compile_definitions("${test_name}" PRIVATE SONN_SOURCE_ROOT="${CMAKE_SOURCE_DIR}" SONN_RESOURCE_ROOT="${SONN_RUNTIME}")
  add_test(NAME "${test_name}" COMMAND "${test_name}" "${CMAKE_SOURCE_DIR}" "${SONN_RUNTIME}")
endforeach()
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
add_test(NAME project_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/validate_project.py")

file(GLOB UI_TESTS CONFIGURE_DEPENDS tests/ui/*.cpp)
foreach(test_source ${UI_TESTS})
 get_filename_component(test_name "${test_source}" NAME_WE)
 add_executable("${test_name}" "${test_source}")
 target_link_libraries("${test_name}" PRIVATE sonn_ui)
 if(MSVC)
   target_compile_options("${test_name}" PRIVATE /utf-8)
 endif()
 add_test(NAME "${test_name}" COMMAND "${test_name}" "${SONN_RUNTIME}")
endforeach()
