cmake_minimum_required(VERSION 3.24)
project(Sonnheide VERSION 0.1.0 LANGUAGES CXX)
option(SONNHEIDE_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
add_library(sonnheide_kernel engine/src/world.cpp engine/src/simulation.cpp engine/src/persistence.cpp)
target_include_directories(sonnheide_kernel PUBLIC engine/include)
target_compile_features(sonnheide_kernel PUBLIC cxx_std_20)
set_target_properties(sonnheide_kernel PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(sonnheide_kernel PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(sonnheide_kernel PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(SONNHEIDE_SANITIZERS)
    target_compile_options(sonnheide_kernel PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(sonnheide_kernel PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(sonnheide_headless apps/headless/main.cpp)
target_link_libraries(sonnheide_headless PRIVATE sonnheide_kernel)
include(CTest)
if(BUILD_TESTING)
  add_executable(sonnheide_kernel_tests tests/kernel_tests.cpp)
  target_link_libraries(sonnheide_kernel_tests PRIVATE sonnheide_kernel)
  add_test(NAME kernel_invariants COMMAND sonnheide_kernel_tests)
  add_executable(sonnheide_allocation_tests tests/transaction_allocation_tests.cpp)
  target_link_libraries(sonnheide_allocation_tests PRIVATE sonnheide_kernel)
  add_test(NAME atomic_allocation_failures COMMAND sonnheide_allocation_tests)
  add_test(NAME headless_reclamation_port_demo COMMAND sonnheide_headless)
endif()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND AND BUILD_TESTING)
  add_test(NAME project_content_contracts COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/validate_project.py)
endif()
