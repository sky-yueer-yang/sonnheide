cmake_minimum_required(VERSION 3.24)
project(Sonnheide VERSION 0.1.0 LANGUAGES CXX)
option(SONNHEIDE_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
add_library(sonnheide_kernel engine/src/world.cpp engine/src/simulation.cpp engine/src/persistence.cpp engine/src/site.cpp engine/src/clothing.cpp)
target_include_directories(sonnheide_kernel PUBLIC engine/include)
add_library(sonnheide_ocean_math presentation/src/ocean_fft.cpp)
target_include_directories(sonnheide_ocean_math PUBLIC presentation/include)
add_library(sonnheide_interaction engine/src/interaction.cpp)
target_include_directories(sonnheide_interaction PUBLIC engine/include)
find_package(Threads REQUIRED)
target_link_libraries(sonnheide_interaction PUBLIC Threads::Threads)
foreach(target sonnheide_kernel sonnheide_ocean_math sonnheide_interaction)
  target_compile_features(${target} PUBLIC cxx_std_20)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(SONNHEIDE_SANITIZERS)
      target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
add_executable(sonnheide_headless apps/headless/main.cpp)
target_link_libraries(sonnheide_headless PRIVATE sonnheide_kernel)
include(CTest)

  target_link_libraries(sonnheide_kernel_tests PRIVATE sonnheide_kernel)

  add_executable(sonnheide_allocation_tests tests/transaction_allocation_tests.cpp)
  target_link_libraries(sonnheide_allocation_tests PRIVATE sonnheide_kernel)

  add_executable(sonnheide_site_tests tests/site_tests.cpp)
  target_link_libraries(sonnheide_site_tests PRIVATE sonnheide_kernel)
  add_test(NAME site_geometry COMMAND sonnheide_site_tests)

  target_link_libraries(sonnheide_clothing_tests PRIVATE sonnheide_kernel)
  add_test(NAME clothing_economy COMMAND sonnheide_clothing_tests)
  add_executable(sonnheide_ocean_fft_tests tests/ocean_fft_tests.cpp)
  target_link_libraries(sonnheide_ocean_fft_tests PRIVATE sonnheide_ocean_math)
  add_test(NAME ocean_fft_reference COMMAND sonnheide_ocean_fft_tests)
